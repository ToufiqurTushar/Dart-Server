5. SERVERPOD BACKEND SERVER COMMANDS
--------------------------------------------------------------------------------
# A. Create New Serverpod Project (Includes Server, Client & Flutter Web/Mobile App):
serverpod create <project_name>

# Example:
serverpod create my_app
# (This automatically creates: my_app_server, my_app_client, and my_app_flutter)

# B. Local Development:
cd my_app_server

# 1. Start Docker Containers (PostgreSQL & Redis):
docker compose up -d

# 2. Generate Serverpod Code (Run whenever models/endpoints change):
serverpod generate

# 3. Create Database Migration (Run whenever models change):
serverpod create-migration

# 4. Start Serverpod Server with Migrations:
dart bin/main.dart --apply-migrations

# C. VPS Production Deployment (Ubuntu):
# Install Dart SDK on Ubuntu VPS:
sudo apt update
sudo apt install apt-transport-https
wget -qO- https://dl-ssl.google.com/linux/linux_signing_key.pub | sudo gpg --dearmor -o /usr/share/keyrings/dart.gpg
echo 'deb [signed-by=/usr/share/keyrings/dart.gpg] https://storage.googleapis.com/download.dartlang.org/linux/debian stable main' | sudo tee /etc/apt/sources.list.d/dart_stable.list
sudo apt update
sudo apt install dart -y

# Run Serverpod in Production Mode on VPS:
cd /var/www/Dart-Server/my_app_server
dart bin/main.dart --mode production --apply-migrations

# Run Serverpod as Background Daemon on VPS (nohup):
cd /var/www/Dart-Server/my_app_server && nohup dart bin/main.dart --mode production --apply-migrations > /dev/null 2>&1 &

# Check running Serverpod process:
ps aux | grep main.dart

# Stop running Serverpod server on VPS:
pkill -f "dart bin/main.dart"


6. FLUTTER WEB APP COMMANDS
--------------------------------------------------------------------------------
# A. Enable Web Support & Run Locally:
cd my_app_flutter
flutter config --enable-web
flutter run -d chrome

# B. Build Web App & Bundle with Serverpod Server:
# Windows (Build & Copy to Serverpod Web directory):
cd my_app_flutter
if (!(Test-Path "..\my_app_server\web\app")) { New-Item -ItemType Directory -Force -Path "..\my_app_server\web\app" }
flutter build web --release
Copy-Item -Path "build\web\*" -Destination "..\my_app_server\web\app" -Recurse -Force

# Linux / VPS Web Build for Serverpod:
cd my_app_flutter
mkdir -p ../my_app_server/web/app
flutter build web --release --base-href /app/ --wasm
cp -r build/web/* ../my_app_server/web/app/

--------------------------------------------------------------------------------
